local function new(class, props, parent)
	local obj = Instance.new(class)

	for k, v in pairs(props or {}) do
		obj[k] = v
	end

	obj.Parent = parent
	return obj
end

local function corner(obj, radius)
	new("UICorner", {
		CornerRadius = UDim.new(0, radius or 10)
	}, obj)
end

local function outline(obj, color, transparency)
	return new("UIStroke", {
		Color = color or T.Accent,
		Transparency = transparency or 0,
		Thickness = 1,
	}, obj)
end

local function animate(obj, duration, props)
	local tw = TweenService:Create(
		obj,
		TweenInfo.new(duration, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
		props
	)
	tw:Play()
	return tw
end

local function label(parent, text, size, position, color)
	return new("TextLabel", {
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = color or T.Text,
		TextSize = 13,
		Font = Enum.Font.GothamMedium,
		TextXAlignment = Enum.TextXAlignment.Left,
		Size = size,
		Position = position,
	}, parent)
end

local function makeDraggable(frame, handle)
	local dragging = false
	local startPos
	local startInput

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			startInput = input.Position
			startPos = frame.Position
		end
	end)

	UIS.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local delta = input.Position - startInput

			frame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)

	UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)
end

makeDraggable(Main, Top)

local CrosshairGui = new("Frame", {
	Name = "CustomCrosshair",
	Size = UDim2.fromOffset(100, 100),
	Position = UDim2.new(0.5, -50, 0.5, -50),
	BackgroundTransparency = 1,

local function rebuildCrosshair()
	CrosshairGui:ClearAllChildren()

	local s = C.CrosshairSize
	local t = C.CrosshairThickness
	local g = C.CrosshairGap
	local color = C.CrosshairColor
	local tr = C.CrosshairTransparency

	local function part(name, size, pos, rotation)
		if C.CrosshairOutline then
			local border = new("Frame", {
				Name = name .. "Outline",
				Size = UDim2.fromOffset(size.X.Offset + 2, size.Y.Offset + 2),
				Position = pos - UDim2.fromOffset(1, 1),
				BackgroundColor3 = Color3.new(0, 0, 0),
				BackgroundTransparency = tr,
				BorderSizePixel = 0,
				Rotation = rotation or 0,
			}, CrosshairGui)

			corner(border, 5)
		end

		local line = new("Frame", {
			Name = name,
			Size = size,
			Position = pos,
			BackgroundColor3 = color,
			BackgroundTransparency = tr,
			BorderSizePixel = 0,
			Rotation = rotation or 0,
		}, CrosshairGui)

		corner(line, 5)
	end

	if C.CrosshairStyle == "Classic" then
		part("Top", UDim2.fromOffset(t, s), UDim2.fromOffset(50-t/2, 50-g-s))
		part("Bottom", UDim2.fromOffset(t, s), UDim2.fromOffset(50-t/2, 50+g))
		part("Left", UDim2.fromOffset(s, t), UDim2.fromOffset(50-g-s, 50-t/2))
		part("Right", UDim2.fromOffset(s, t), UDim2.fromOffset(50+g, 50-t/2))

	elseif C.CrosshairStyle == "Circle" then
		local circle = new("Frame", {
			Size = UDim2.fromOffset(s*2, s*2),
			Position = UDim2.fromOffset(50-s, 50-s),
			BackgroundTransparency = 1,
		}, CrosshairGui)

		new("UICorner", {CornerRadius = UDim.new(1, 0)}, circle)
		new("UIStroke", {
			Color = color,
			Thickness = t,
			Transparency = tr,
		}, circle)

	elseif C.CrosshairStyle == "Dot" then
		local dot = new("Frame", {
			Size = UDim2.fromOffset(t*2, t*2),
			Position = UDim2.fromOffset(50-t, 50-t),
			BackgroundColor3 = color,
			BackgroundTransparency = tr,
		}, CrosshairGui)
		corner(dot, 20)

	elseif C.CrosshairStyle == "Square" then
		local square = new("Frame", {
			Size = UDim2.fromOffset(s*2, s*2),
			Position = UDim2.fromOffset(50-s, 50-s),
			BackgroundTransparency = 1,
		}, CrosshairGui)
		outline(square, color, tr)

	elseif C.CrosshairStyle == "X" then
		part("X1", UDim2.fromOffset(s*2, t), UDim2.fromOffset(50-s, 50-t/2), 45)
		part("X2", UDim2.fromOffset(s*2, t), UDim2.fromOffset(50-s, 50-t/2), -45)
	end

	if C.CrosshairDot and C.CrosshairStyle ~= "Dot" then
		local dot = new("Frame", {
			Size = UDim2.fromOffset(t, t),
			Position = UDim2.fromOffset(50-t/2, 50-t/2),
			BackgroundColor3 = color,
			BackgroundTransparency = tr,
		}, CrosshairGui)
		corner(dot, 20)
	end

	CrosshairGui.Visible = C.Crosshair
end

rebuildCrosshair()

		Size = UDim2.fromOffset(13, 13),
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new((value-min)/(max-min), 0, 0.5, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
	}, bar)
	corner(knob, 20)

	local active = false

	local function setValue(x)
		local p = math.clamp(
			(x-bar.AbsolutePosition.X)/bar.AbsoluteSize.X,
			0, 1
		)

		value = math.floor(min+(max-min)*p+0.5)
		fill.Size = UDim2.new(p, 0, 1, 0)
		knob.Position = UDim2.new(p, 0, 0.5, 0)
		valueLabel.Text = tostring(value)
		callback(value)
	end

	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			active = true
			setValue(input.Position.X)
		end
	end)

	UIS.InputChanged:Connect(function(input)
		if active and input.UserInputType == Enum.UserInputType.MouseMovement then
			setValue(input.Position.X)
		end
	end)

	UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			active = false
		end
	end)
end

local ESPData = {}

local function removeESP(player)
	local data = ESPData[player]
	if not data then return end

	if data.Highlight then data.Highlight:Destroy() end
	if data.Billboard then data.Billboard:Destroy() end

	ESPData[player] = nil
end

local function updateESP(player)
	if player == LP then return end

	local character = player.Character
	if not character then
		removeESP(player)
		return
	end

	local head = character:FindFirstChild("Head")
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")

	if not head or not humanoid or not root then return end

	local data = ESPData[player] or {}
	ESPData[player] = data

	if not data.Highlight then
		data.Highlight = new("Highlight", {
			FillTransparency = 0.8,
			OutlineTransparency = 0,
			DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
		}, character)
	end

	data.Highlight.Adornee = character
	data.Highlight.Enabled = C.ESP
	data.Highlight.OutlineColor = T.Accent
	data.Highlight.FillColor = Color3.fromRGB(180, 80, 255)

	if not data.Billboard then
		data.Billboard = new("BillboardGui", {
			Size = UDim2.fromOffset(160, 65),
			StudsOffset = Vector3.new(0, 3, 0),
			AlwaysOnTop = true,
			Adornee = head,
		}, character)

		data.Text = new("TextLabel", {
			Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 1,
			TextColor3 = Color3.new(1, 1, 1),
			TextStrokeTransparency = 0.3,
			Font = Enum.Font.GothamBold,
			TextSize = 12,
		}, data.Billboard)
	end

	local lines = {}

	if C.Names then table.insert(lines, player.DisplayName) end
	if C.Health then table.insert(lines, "HP: " .. math.floor(humanoid.Health)) end

	if C.Distance then
		local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
		if myRoot then
			table.insert(lines, math.floor((myRoot.Position-root.Position).Magnitude) .. " studs")
		end
	end

	data.Text.Text = table.concat(lines, "\n")
	data.Billboard.Enabled = C.ESP and #lines > 0
end

local AimHeld = false
local LockedTarget = nil

local function isFriend(player)
	if not C.IgnoreFriends then return false end

	local success, result = pcall(function()
		return LP:IsFriendsWith(player.UserId)
	end)

	return success and result
end

local function findTarget()
	local best = nil
	local bestDistance = C.AimFOV

	local center = Vector2.new(
		Camera.ViewportSize.X/2,
		Camera.ViewportSize.Y/2
	)

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LP and not isFriend(player) then
			local character = player.Character
			local head = character and character:FindFirstChild("Head")
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")

			if head and humanoid and humanoid.Health > 0 then
				local pos, visible = Camera:WorldToViewportPoint(head.Position)

				if visible and pos.Z > 0 then
					local distance = (
						Vector2.new(pos.X, pos.Y)-center
					).Magnitude

					if distance < bestDistance then
						bestDistance = distance
						best = head
					end
				end
			end
		end
	end

	return best
end

local function validTarget(part)
	if not part or not part.Parent then return false end

	local character = part:FindFirstAncestorOfClass("Model")
	if not character then return false end

	local player = Players:GetPlayerFromCharacter(character)
	local humanoid = character:FindFirstChildOfClass("Humanoid")

	return player ~= nil and player ~= LP
		and humanoid ~= nil and humanoid.Health > 0
end

	local character = LP.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")

	if not root or not humanoid then return end

	stopFly()
	Flying = true
	humanoid.PlatformStand = true

	FlyVelocity = Instance.new("BodyVelocity")
	FlyVelocity.MaxForce = Vector3.new(1e6, 1e6, 1e6)
	FlyVelocity.Velocity = Vector3.zero
	FlyVelocity.Parent = root

	FlyGyro = Instance.new("BodyGyro")
	FlyGyro.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
	FlyGyro.P = 1e5
	FlyGyro.CFrame = Camera.CFrame
	FlyGyro.Parent = root

	FlyConnection = RunService.RenderStepped:Connect(function()
		if not root.Parent then
			stopFly()
			return
		end

		local dir = Vector3.zero

		if FlyKeys.W then dir += Camera.CFrame.LookVector end
		if FlyKeys.S then dir -= Camera.CFrame.LookVector end
		if FlyKeys.A then dir -= Camera.CFrame.RightVector end
		if FlyKeys.D then dir += Camera.CFrame.RightVector end
		if FlyKeys.Space then dir += Vector3.yAxis end
		if FlyKeys.LeftControl then dir -= Vector3.yAxis end

		if dir.Magnitude > 0 then dir = dir.Unit end

		FlyVelocity.Velocity = dir * C.FlySpeed
		FlyGyro.CFrame = Camera.CFrame
	end)
end

local OriginalCollision = {}
local NoClipConnection

local function setNoClip(enabled)
	C.NoClip = enabled

	if NoClipConnection then
		NoClipConnection:Disconnect()
		NoClipConnection = nil
	end

	if not enabled then
		for part, value in pairs(OriginalCollision) do
			if part and part.Parent then part.CanCollide = value end
		end

		table.clear(OriginalCollision)
		return
	end

	NoClipConnection = RunService.Stepped:Connect(function()
		local character = LP.Character
		if not character then return end

		for _, part in ipairs(character:GetDescendants()) do
			if part:IsA("BasePart") then
				if OriginalCollision[part] == nil then
					OriginalCollision[part] = part.CanCollide
				end

				part.CanCollide = false
			end
		end
	end)
end

local function matchesAim(input)
	if C.AimKey == Enum.UserInputType.MouseButton2 then
		return input.UserInputType == Enum.UserInputType.MouseButton2
	end

	return input.KeyCode == C.AimKey
end

UIS.InputBegan:Connect(function(input, processed)
	if input.KeyCode == C.MenuKey then
		toggleMenu()
		return
	end

	if processed then return end

	if matchesAim(input) and C.AimEnabled then
		AimHeld = true
		LockedTarget = findTarget()
	end

	if input.KeyCode == Enum.KeyCode.W then FlyKeys.W = true end
	if input.KeyCode == Enum.KeyCode.A then FlyKeys.A = true end
	if input.KeyCode == Enum.KeyCode.S then FlyKeys.S = true end
	if input.KeyCode == Enum.KeyCode.D then FlyKeys.D = true end
	if input.KeyCode == Enum.KeyCode.Space then FlyKeys.Space = true end
	if input.KeyCode == Enum.KeyCode.LeftControl then FlyKeys.LeftControl = true end
end)

UIS.InputEnded:Connect(function(input)
	if matchesAim(input) then
		AimHeld = false
		LockedTarget = nil
	end

	if input.KeyCode == Enum.KeyCode.W then FlyKeys.W = false end
	if input.KeyCode == Enum.KeyCode.A then FlyKeys.A = false end
	if input.KeyCode == Enum.KeyCode.S then FlyKeys.S = false end
	if input.KeyCode == Enum.KeyCode.D then FlyKeys.D = false end
	if input.KeyCode == Enum.KeyCode.Space then FlyKeys.Space = false end
	if input.KeyCode == Enum.KeyCode.LeftControl then FlyKeys.LeftControl = false end
end)

RunService.RenderStepped:Connect(function(dt)
	Camera = workspace.CurrentCamera

	CrosshairGui.Visible = C.Crosshair
	FOV.Visible = C.ShowFOV

	if C.ESP then
		for _, player in ipairs(Players:GetPlayers()) do
			updateESP(player)
		end
	end

	if AimHeld and C.AimEnabled and not Main.Visible then
		if validTarget(LockedTarget) then
			local targetCFrame = CFrame.lookAt(
				Camera.CFrame.Position,
				LockedTarget.Position
			)

			local alpha = 1-math.exp(-C.AimSmooth*dt)

			Camera.CFrame = Camera.CFrame:Lerp(
				targetCFrame,
				math.clamp(alpha, 0, 1)
			)
		else
			AimHeld = false
			LockedTarget = nil
		end
	end
end)

	end
